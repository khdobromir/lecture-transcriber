cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "In-source builds are disabled. Use cmake -S . -B out/build/<name>.")
endif()
project(transcribe VERSION 0.4.1 DESCRIPTION "Транскрипция русской речи на CPU" LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Windows)$")
    message(FATAL_ERROR "Supported platforms: Linux and Windows")
endif()
option(TRANSCRIBE_BUILD_GUI "Build the Qt Quick desktop application" OFF)
option(TRANSCRIBE_WARNINGS_AS_ERRORS "Treat project compiler warnings as errors" OFF)
function(transcribe_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive-)
        # Import-time DLL lookup also excludes the caller's cwd and PATH.
        target_link_options(${target} PRIVATE /DEPENDENTLOADFLAG:0xA00)
        target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        if(TRANSCRIBE_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(TRANSCRIBE_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()
if(WIN32)
    enable_language(RC)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    set(TRANSCRIBE_PLATFORM src/platform_windows.cpp src/process_windows.cpp)
else()
    set(TRANSCRIBE_PLATFORM src/platform_linux.cpp src/process.cpp src/process_guard_linux.cpp)
endif()
add_library(transcribe_core STATIC ${TRANSCRIBE_PLATFORM} src/platform.cpp src/cancellation.cpp
    src/audio.cpp src/cli.cpp src/result.cpp src/progress.cpp src/media.cpp src/pipeline.cpp src/protocol.cpp)
target_compile_features(transcribe_core PUBLIC cxx_std_23)
target_include_directories(transcribe_core PUBLIC src)
target_include_directories(transcribe_core SYSTEM PUBLIC third_party)
set_target_properties(transcribe_core PROPERTIES CXX_EXTENSIONS OFF)
transcribe_warnings(transcribe_core)
if(WIN32)
    target_link_libraries(transcribe_core PUBLIC advapi32)
endif()
add_executable(transcribe src/main.cpp)
target_link_libraries(transcribe PRIVATE transcribe_core)
target_compile_features(transcribe PRIVATE cxx_std_23)
set_target_properties(transcribe PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(transcribe PRIVATE TRANSCRIBE_VERSION="${PROJECT_VERSION}")
transcribe_warnings(transcribe)
if(MINGW)
    target_link_options(transcribe PRIVATE -municode)
endif()
if(WIN32)
    target_sources(transcribe PRIVATE packaging/windows/transcribe-icon.rc)
    target_include_directories(transcribe PRIVATE packaging/windows)
    if(MSVC)
        target_sources(transcribe PRIVATE packaging/windows/transcribe.manifest)
    else()
        target_sources(transcribe PRIVATE packaging/windows/transcribe.manifest.rc)
        target_include_directories(transcribe PRIVATE packaging/windows)
    endif()
endif()
install(TARGETS transcribe RUNTIME DESTINATION bin)

install(FILES scripts/models.tsv DESTINATION share/transcribe)
include(CTest)
if(TRANSCRIBE_BUILD_GUI)
    if(CMAKE_VERSION VERSION_LESS 3.24)
        message(FATAL_ERROR "The GUI requires CMake 3.24 or newer")
    endif()
    add_subdirectory(gui)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
