cmake_minimum_required(VERSION 3.20)
project(transcribe VERSION 0.3.0 DESCRIPTION "Транскрипция русской речи на CPU" LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Эта версия предназначена для Linux (Omarchy/Arch)")
endif()
add_library(transcribe_core STATIC src/process.cpp src/audio.cpp src/cli.cpp src/result.cpp src/progress.cpp src/media.cpp)
target_compile_features(transcribe_core PUBLIC cxx_std_23)
target_include_directories(transcribe_core PUBLIC src)
set_target_properties(transcribe_core PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(transcribe_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(transcribe src/main.cpp)
target_link_libraries(transcribe PRIVATE transcribe_core)
target_compile_features(transcribe PRIVATE cxx_std_23)
set_target_properties(transcribe PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(transcribe PRIVATE TRANSCRIBE_VERSION="${PROJECT_VERSION}")
target_compile_options(transcribe PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS transcribe RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
    find_program(FFMPEG_EXECUTABLE ffmpeg REQUIRED)
    add_executable(mock_tools tests/mock_tools.cpp)
    add_executable(smoke_real tests/smoke_real.cpp)
    add_executable(benchmark tests/benchmark.cpp)
    foreach(target mock_tools smoke_real benchmark)
        target_compile_features(${target} PRIVATE cxx_std_23)
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endforeach()
    target_compile_definitions(smoke_real PRIVATE
        TRANSCRIBE_BINARY="$<TARGET_FILE:transcribe>" PROJECT_SOURCE="${PROJECT_SOURCE_DIR}")
    target_compile_definitions(benchmark PRIVATE TRANSCRIBE_BINARY="$<TARGET_FILE:transcribe>")
    target_link_libraries(smoke_real PRIVATE transcribe_core)
    target_link_libraries(benchmark PRIVATE transcribe_core)
    add_dependencies(smoke_real transcribe)
    add_dependencies(benchmark transcribe)
    foreach(suite pipeline installation smoke_real)
        add_executable(test_${suite} tests/test_${suite}.cpp)
        target_compile_features(test_${suite} PRIVATE cxx_std_23)
        set_target_properties(test_${suite} PROPERTIES CXX_EXTENSIONS OFF)
        target_compile_options(test_${suite} PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_definitions(test_${suite} PRIVATE
            TRANSCRIBE_BINARY="$<TARGET_FILE:transcribe>" MOCK_BINARY="$<TARGET_FILE:mock_tools>"
            SMOKE_BINARY="$<TARGET_FILE:smoke_real>" PROJECT_SOURCE="${PROJECT_SOURCE_DIR}"
            TRANSCRIBE_VERSION="${PROJECT_VERSION}")
        add_dependencies(test_${suite} transcribe mock_tools smoke_real)
    endforeach()
    target_link_libraries(test_pipeline PRIVATE transcribe_core)
    add_test(NAME pipeline COMMAND test_pipeline)
    add_test(NAME installation COMMAND test_installation)
    add_test(NAME smoke_preflight COMMAND test_smoke_real)
    set_tests_properties(pipeline installation smoke_preflight PROPERTIES TIMEOUT 120)
    foreach(suite cli audio features)
        add_executable(test_unit_${suite} tests/test_unit_${suite}.cpp)
        target_link_libraries(test_unit_${suite} PRIVATE transcribe_core)
        target_compile_options(test_unit_${suite} PRIVATE -Wall -Wextra -Wpedantic)
        add_test(NAME unit_${suite} COMMAND test_unit_${suite})
        set_tests_properties(unit_${suite} PROPERTIES TIMEOUT 30 LABELS unit)
    endforeach()
    set_tests_properties(pipeline PROPERTIES LABELS integration)
    set_tests_properties(installation PROPERTIES LABELS installation)
    set_tests_properties(smoke_preflight PROPERTIES LABELS preflight)
endif()
