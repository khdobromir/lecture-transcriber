cmake_minimum_required(VERSION 3.20)
project(transcribe VERSION 0.3.0 DESCRIPTION "Транскрипция русской речи на CPU" LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Windows)$")
    message(FATAL_ERROR "Supported platforms: Linux and Windows")
endif()
option(TRANSCRIBE_BUILD_GUI "Build the Qt Quick desktop application" OFF)
option(TRANSCRIBE_WARNINGS_AS_ERRORS "Treat project compiler warnings as errors" OFF)
function(transcribe_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive-)
        target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        if(TRANSCRIBE_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(TRANSCRIBE_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()
if(WIN32)
    enable_language(RC)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    set(TRANSCRIBE_PLATFORM src/platform_windows.cpp src/process_windows.cpp)
else()
    set(TRANSCRIBE_PLATFORM src/platform_linux.cpp src/process.cpp)
endif()
add_library(transcribe_core STATIC ${TRANSCRIBE_PLATFORM} src/platform.cpp src/cancellation.cpp
    src/audio.cpp src/cli.cpp src/result.cpp src/progress.cpp src/media.cpp src/pipeline.cpp src/protocol.cpp)
target_compile_features(transcribe_core PUBLIC cxx_std_23)
target_include_directories(transcribe_core PUBLIC src)
target_include_directories(transcribe_core SYSTEM PUBLIC third_party)
set_target_properties(transcribe_core PROPERTIES CXX_EXTENSIONS OFF)
transcribe_warnings(transcribe_core)
if(WIN32)
    target_link_libraries(transcribe_core PUBLIC advapi32)
endif()
add_executable(transcribe src/main.cpp)
target_link_libraries(transcribe PRIVATE transcribe_core)
target_compile_features(transcribe PRIVATE cxx_std_23)
set_target_properties(transcribe PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(transcribe PRIVATE TRANSCRIBE_VERSION="${PROJECT_VERSION}")
transcribe_warnings(transcribe)
if(MINGW)
    target_link_options(transcribe PRIVATE -municode)
endif()
if(WIN32)
    target_sources(transcribe PRIVATE packaging/windows/transcribe.manifest.rc)
    target_include_directories(transcribe PRIVATE packaging/windows)
endif()
install(TARGETS transcribe RUNTIME DESTINATION bin)

install(FILES scripts/models.tsv DESTINATION share/transcribe)
include(CTest)
if(TRANSCRIBE_BUILD_GUI)
    if(CMAKE_VERSION VERSION_LESS 3.24)
        message(FATAL_ERROR "The GUI requires CMake 3.24 or newer")
    endif()
    add_subdirectory(gui)
endif()

    add_executable(mock_portable tests/mock_portable.cpp)
    target_link_libraries(mock_portable PRIVATE transcribe_core)
    transcribe_warnings(mock_portable)
    if(MINGW)
        target_link_options(mock_portable PRIVATE -municode)
    endif()
    add_test(NAME machine_protocol COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_machine.py $<TARGET_FILE:transcribe> $<TARGET_FILE:mock_portable>)
    set_tests_properties(machine_protocol PROPERTIES TIMEOUT 90 LABELS integration)
    add_executable(test_portable tests/test_portable.cpp)
    target_link_libraries(test_portable PRIVATE transcribe_core)
    transcribe_warnings(test_portable)
    if(MINGW)
        target_link_options(test_portable PRIVATE -municode)
    endif()
    add_test(NAME portable_platform COMMAND test_portable)
    set_tests_properties(portable_platform PROPERTIES TIMEOUT 30 LABELS integration)

if(BUILD_TESTING AND NOT WIN32)
    find_program(FFMPEG_EXECUTABLE ffmpeg REQUIRED)
    add_executable(mock_tools tests/mock_tools.cpp)
    add_executable(smoke_real tests/smoke_real.cpp)
    add_executable(benchmark tests/benchmark.cpp)
    foreach(target mock_tools smoke_real benchmark)
        target_compile_features(${target} PRIVATE cxx_std_23)
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
        transcribe_warnings(${target})
    endforeach()
    target_compile_definitions(smoke_real PRIVATE
        TRANSCRIBE_BINARY="$<TARGET_FILE:transcribe>" PROJECT_SOURCE="${PROJECT_SOURCE_DIR}")
    target_compile_definitions(benchmark PRIVATE TRANSCRIBE_BINARY="$<TARGET_FILE:transcribe>")
    target_link_libraries(smoke_real PRIVATE transcribe_core)
    target_link_libraries(benchmark PRIVATE transcribe_core)
    add_dependencies(smoke_real transcribe)
    add_dependencies(benchmark transcribe)
    foreach(suite pipeline installation smoke_real)
        add_executable(test_${suite} tests/test_${suite}.cpp)
        target_compile_features(test_${suite} PRIVATE cxx_std_23)
        set_target_properties(test_${suite} PROPERTIES CXX_EXTENSIONS OFF)
        transcribe_warnings(test_${suite})
        target_compile_definitions(test_${suite} PRIVATE
            TRANSCRIBE_BINARY="$<TARGET_FILE:transcribe>" MOCK_BINARY="$<TARGET_FILE:mock_tools>"
            SMOKE_BINARY="$<TARGET_FILE:smoke_real>" PROJECT_SOURCE="${PROJECT_SOURCE_DIR}"
            TRANSCRIBE_VERSION="${PROJECT_VERSION}")
        add_dependencies(test_${suite} transcribe mock_tools smoke_real)
    endforeach()
    target_link_libraries(test_pipeline PRIVATE transcribe_core)
    add_test(NAME pipeline COMMAND test_pipeline)
    add_test(NAME installation COMMAND test_installation)
    add_test(NAME smoke_preflight COMMAND test_smoke_real)
    set_tests_properties(pipeline installation smoke_preflight PROPERTIES TIMEOUT 120)
    foreach(suite cli audio features)
        add_executable(test_unit_${suite} tests/test_unit_${suite}.cpp)
        target_link_libraries(test_unit_${suite} PRIVATE transcribe_core)
        transcribe_warnings(test_unit_${suite})
        add_test(NAME unit_${suite} COMMAND test_unit_${suite})
        set_tests_properties(unit_${suite} PROPERTIES TIMEOUT 30 LABELS unit)
    endforeach()
    set_tests_properties(pipeline PROPERTIES LABELS integration)
    set_tests_properties(installation PROPERTIES LABELS installation)
    set_tests_properties(smoke_preflight PROPERTIES LABELS preflight)
endif()
