cmake_minimum_required(VERSION 3.20)
project(vklecture LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Эта версия предназначена для Linux (Omarchy/Arch)")
endif()
add_executable(vklecture src/main.cpp)
target_compile_features(vklecture PRIVATE cxx_std_23)
target_compile_options(vklecture PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS vklecture RUNTIME DESTINATION bin)
